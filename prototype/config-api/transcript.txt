1. Defaults

> fix_text(x)

[B  config object only]
> fix_text(x)
   fix_text ran with all defaults

[C  both: named arguments override a config]
> fix_text(x)
   fix_text ran with all defaults

2. Keep curly quotes
======================================================================

[A  plain arguments only]
> fix_text(x, uncurl_quotes = FALSE)
   fix_text ran with uncurl_quotes = FALSE

[B  config object only]
> fix_text(x, ftfy_config(uncurl_quotes = FALSE))
   fix_text ran with uncurl_quotes = FALSE

[C  both: named arguments override a config]
> fix_text(x, uncurl_quotes = FALSE)
   fix_text ran with uncurl_quotes = FALSE

======================================================================
3. Same cautious settings for two calls
======================================================================

[A  plain arguments only]
> fix_text(x, uncurl_quotes = FALSE, fix_character_width = FALSE, normalization = NULL)
   fix_text ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL
> fix_and_explain(y, uncurl_quotes = FALSE, fix_character_width = FALSE, normalization = NULL)
   fix_and_explain ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL

[B  config object only]
> cautious <- ftfy_config(uncurl_quotes = FALSE, fix_character_width = FALSE, normalization = NULL)
> fix_text(x, cautious)
   fix_text ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL
> fix_and_explain(y, cautious)
   fix_and_explain ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL

[C  both: named arguments override a config]
> cautious <- ftfy_config(uncurl_quotes = FALSE, fix_character_width = FALSE, normalization = NULL)
> fix_text(x, config = cautious)
   fix_text ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL
> fix_and_explain(y, config = cautious)
   fix_and_explain ran with fix_character_width = FALSE, uncurl_quotes = FALSE, normalization = NULL

======================================================================
4. Saved settings, one change for one call
======================================================================

[A  plain arguments only]
> # no saved settings to start from; spell everything out again
> fix_text(x, uncurl_quotes = FALSE, fix_character_width = FALSE, unescape_html = FALSE)
   fix_text ran with unescape_html = FALSE, fix_character_width = FALSE, uncurl_quotes = FALSE

[B  config object only]
> cautious <- ftfy_config(uncurl_quotes = FALSE, fix_character_width = FALSE)
> # no way to tweak a config; build a second one
> fix_text(x, ftfy_config(uncurl_quotes = FALSE, fix_character_width = FALSE, unescape_html = FALSE))
   fix_text ran with unescape_html = FALSE, fix_character_width = FALSE, uncurl_quotes = FALSE

[C  both: named arguments override a config]
> cautious <- ftfy_config(uncurl_quotes = FALSE, fix_character_width = FALSE)
> fix_text(x, unescape_html = FALSE, config = cautious)
   fix_text ran with unescape_html = FALSE, fix_character_width = FALSE, uncurl_quotes = FALSE
